import urllib.request
import json

# Paste the extracted comma-separated lists here
ASSURED_OSS_FREE = "gremlinpython, grequests, griffe, grip, grpcio...".replace(" ", "").split(",")
ASSURED_OSS_PREMIUM = "APScheduler, Adafruit-Blinka, Adyen, Alarmageddon...".replace(" ", "").split(",")

def generate_enterprise_manifest(assured_free, assured_premium, total_limit=5000):
    """
    Merges Google Assured OSS packages with the top conda-forge ecosystem 
    to create a hardened 5000-package environment limit.
    """
    print("Initializing Google Assured OSS priority list...")
    
    # Lowercase to match conda-forge naming conventions
    priority_packages = set([pkg.lower() for pkg in assured_free + assured_premium])
    
    print(f"Fetching repodata from conda-forge...")
    url = "https://conda.anaconda.org/conda-forge/linux-64/repodata.json"
    
    try:
        req = urllib.request.Request(url, headers={'User-Agent': 'Mozilla/5.0'})
        with urllib.request.urlopen(req) as response:
            data = json.loads(response.read().decode('utf-8'))
            
        packages = data.get("packages", {})
        
        # Extract unique conda-forge packages
        forge_packages = set(pkg_info["name"].lower() for pkg_info in packages.values())
        
        # 1. Guarantee Google Assured packages are included (Intersection with conda-forge)
        secured_layer = priority_packages.intersection(forge_packages)
        
        # 2. Fill the remaining spots with top conda-forge general packages
        remaining_slots = total_limit - len(secured_layer)
        general_layer = list(forge_packages.difference(secured_layer))[:remaining_slots]
        
        # Final Curated List
        final_manifest = list(secured_layer) + general_layer
        
        print(f"Manifest complete: {len(secured_layer)} Secured Packages + {len(general_layer)} General Packages.")
        return sorted(final_manifest)
        
    except Exception as e:
        print(f"Error executing pipeline: {e}")
        return []
